def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

import { NextRequest, NextResponse } from "next/server";
import {
  getFinanceApiHeaders,
  getFinanceApiUrl,
} from "@/lib/finance-api";

// export const dynamic = "force-dynamic";

type RouteContext = {
  params: Promise<{ expenseId: string }>;
};

function validateExpenseId(id: string): boolean {
  const parsed = Number(id);
  return Number.isSafeInteger(parsed) && parsed > 0;
}

export async function PATCH(
  request: NextRequest,
  { params }: RouteContext
) {
  try {
    const { expenseId } = await params;

    if (!validateExpenseId(expenseId)) {
      return NextResponse.json(
        { detail: "Invalid expense ID." },
        { status: 400 }
      );
    }

    const headers = await getFinanceApiHeaders();
    const body = await request.json();

    const response = await fetch(
      `${getFinanceApiUrl()}/expenses/${expenseId}`,
      {
        method: "PATCH",
        headers,
        body: JSON.stringify(body),
        cache: "no-store",
      }
    );

    const data = await response.text();

    return new NextResponse(data, {
      status: response.status,
      headers: {
        "Content-Type":
          response.headers.get("Content-Type") ?? "application/json",
      },
    });
  } catch (error) {
    if (
      error instanceof Error &&
      error.message === "Authentication required."
    ) {
      return NextResponse.json(
        { detail: "Authentication required." },
        { status: 401 }
      );
    }

    console.error("PATCH /api/expenses failed:", error);

    return NextResponse.json(
      { detail: "Could not update expense." },
      { status: 500 }
    );
  }
}

export async function DELETE(
  _request: NextRequest,
  { params }: RouteContext
) {
  try {
    const { expenseId } = await params;

    if (!validateExpenseId(expenseId)) {

        { status: 400 }
      );

    const headers = await getFinanceApiHeaders();

        method: "DELETE",
        headers,

      }
    );

    const data = await response.text();

    return new NextResponse(data, {
      status: response.status,

        "Content-Type":
          response.headers.get("Content-Type") ?? "application/json",

    });

// export const dynamic = "force-dynamic";

export async function GET(request: NextRequest) {
  try {
    const headers = await getFinanceApiHeaders();

    const query = request.nextUrl.searchParams.toString();
    const url = `${getFinanceApiUrl()}/expenses${query ? `?${query}` : ""}`;

    const response = await fetch(url, {
      method: "GET",
      headers,
      cache: "no-store",
    });

    const data = await response.text();

    return new NextResponse(data, {
      status: response.status,
      headers: {
        "Content-Type":
          response.headers.get("Content-Type") ?? "application/json",
      },
    });
  } catch (error) {
    if (
      error instanceof Error &&
      error.message === "Authentication required."
    ) {
      return NextResponse.json(
        { detail: "Authentication required." },
        { status: 401 }
      );
    }

    console.error("GET /api/expenses failed:", error);

    return NextResponse.json(
      { detail: "Could not retrieve expenses." },
      { status: 500 }
    );
  }
}

export async function POST(request: NextRequest) {
  try {
    const headers = await getFinanceApiHeaders();
    const body = await request.json();

    const response = await fetch(`${getFinanceApiUrl()}/expenses`, {
      method: "POST",
      headers,
      body: JSON.stringify(body),
      cache: "no-store",
    });

  getFinanceApiHeaders,
  getFinanceApiUrl,

const allowedTypes = ["monthly", "categories", "trend"];

type RouteContext = {
  params: Promise<{ summaryType: string }>;
};

export async function GET(
  request: NextRequest,
  { params }: RouteContext
) {
  const { summaryType } = await params;

  if (!allowedTypes.includes(summaryType)) {
    return NextResponse.json(
      { detail: "Summary endpoint not found." },
      { status: 404 }
    );
  }

  try {
    const headers = await getFinanceApiHeaders();

    const query = request.nextUrl.searchParams.toString();

    const url = `${getFinanceApiUrl()}/summary/${summaryType}${
      query ? `?${query}` : ""
    }`;

      method: "GET",
      headers,
      cache: "no-store",
    });

      status: response.status,

    });

      error instanceof Error &&

import { withAuth } from "@workos-inc/authkit-nextjs";
import { SignJWT } from "jose";

const API_URL = process.env.FINANCE_API_URL;
const API_SECRET = process.env.INTERNAL_API_SECRET;

const API_AUDIENCE = "http://localhost:8001";

export async function getFinanceApiHeaders() {
  const { user } = await withAuth();

  if (!user) {
    throw new Error("Authentication required.");
  }

  if (!API_SECRET) {
    throw new Error("Internal API secret is not configured.");
  }

  const secret = new TextEncoder().encode(API_SECRET);

  const token = await new SignJWT({})
    .setProtectedHeader({ alg: "HS256" })
    .setSubject(user.id)
    .setIssuer("finora-web")
    .setAudience(API_AUDIENCE)
    .setIssuedAt()
    .setExpirationTime("5m")
    .sign(secret);

  return {
    Authorization: `Bearer ${token}`,
    "Content-Type": "application/json",
  };
}

export function getFinanceApiUrl() {
  if (!API_URL) {
    throw new Error("Finance API URL is not configured.");
  }

  return API_URL;
}

from datetime import date
from sqlalchemy import func
from sqlalchemy.orm import Session

from DB.models import Expense


def get_monthly_summary(
    db: Session,
    user_id: int,
    month: date,
):
    """
    Returns spending total, transaction count,
    and comparison with the previous month.
    """

    current_start = month.replace(day=1)

    if current_start.month == 12:
        next_start = current_start.replace(
            year=current_start.year + 1,
            month=1,
        )
    else:
        next_start = current_start.replace(
            month=current_start.month + 1,
        )

    previous_start = (
        current_start.replace(day=1)
        if current_start.month > 1
        else current_start.replace(year=current_start.year - 1, month=12)
    )

    if current_start.month == 1:
        previous_start = current_start.replace(
            year=current_start.year - 1,
            month=12,
        )
    else:
        previous_start = current_start.replace(
            month=current_start.month - 1,
        )

    current_total, current_count = (
        db.query(
            func.coalesce(func.sum(Expense.amount), 0),
            func.count(Expense.id),
        )
        .filter(
            Expense.user_id == user_id,
            Expense.date >= current_start,
            Expense.date < next_start,
        )
        .one()
    )

    previous_total = (
        db.query(func.coalesce(func.sum(Expense.amount), 0))
        .filter(
            Expense.user_id == user_id,
            Expense.date >= previous_start,
            Expense.date < current_start,
        )
        .scalar()
    )

    percentage_change = None

    if previous_total > 0:
        percentage_change = (
            (current_total - previous_total) / previous_total
        ) * 100

    return {
        "month": current_start.strftime("%Y-%m"),
        "total_spent": current_total,
        "transaction_count": current_count,
        "previous_month_total": previous_total,
        "percentage_change": percentage_change,
    }


def get_category_summary(
    db: Session,
    user_id: int,
    month: date,
):
    """Returns spending totals grouped by category."""

    month_start = month.replace(day=1)

    if month_start.month == 12:
        next_month = month_start.replace(
            year=month_start.year + 1,
            month=1,
        )
    else:
        next_month = month_start.replace(
            month=month_start.month + 1,
        )

    results = (
        db.query(
            Expense.category,
            func.sum(Expense.amount).label("total"),
            func.count(Expense.id).label("transaction_count"),
        )
        .filter(
            Expense.user_id == user_id,
            Expense.date >= month_start,
            Expense.date < next_month,
        )
        .group_by(Expense.category)
        .order_by(func.sum(Expense.amount).desc())
        .all()
    )

    return [
        {
            "category": category,
            "total": total,
            "transaction_count": count,
        }
        for category, total, count in results
    ]


def get_monthly_trend(
    db: Session,
    user_id: int,
    months: int = 6,
):
    """Returns monthly spending totals for the last N months."""

    today = date.today()
    current_month = today.replace(day=1)

    # Include the current month and the preceding months.
    month_index = current_month.year * 12 + current_month.month - 1
    first_index = month_index - months + 1

    start_year, start_month_index = divmod(first_index, 12)
    start_date = date(start_year, start_month_index + 1, 1)

    results = (
        db.query(
            func.extract("year", Expense.date).label("year"),
            func.extract("month", Expense.date).label("month"),
            func.sum(Expense.amount).label("total"),
            func.count(Expense.id).label("transaction_count"),
        )
        .filter(
            Expense.user_id == user_id,
            Expense.date >= start_date,
            Expense.date < (
                date(current_month.year + 1, 1, 1)
                if current_month.month == 12
                else current_month.replace(month=current_month.month + 1)
            ),
        )
        .group_by(
            func.extract("year", Expense.date),
            func.extract("month", Expense.date),
        )
        .order_by(
            func.extract("year", Expense.date),
            func.extract("month", Expense.date),
        )
        .all()
    )

    # Convert database results into a lookup.
    totals = {
        (int(year), int(month)): {
            "total": total,
            "transaction_count": count,
        }
        for year, month, total, count in results
    }

    # Include months with zero spending.
    trend = []

    for offset in range(months):
        index = first_index + offset
        year, month_index = divmod(index, 12)
        month_number = month_index + 1

        data = totals.get((year, month_number), {})

        trend.append({
            "month": f"{year}-{month_number:02d}",
            "total_spent": data.get("total", 0),
            "transaction_count": data.get("transaction_count", 0),
        })

    return trend

 async function handleSubmit(e: FormEvent<HTMLFormElement>) {

from sqlalchemy.orm import Session
from sqlalchemy.exc import IntegrityError

from DB.models import User


def provision_workos_user(
    db: Session,
    workos_user_id: str,
    email: str,
    name: str,
):
    email = email.strip().lower()
    name = name.strip() or email

    # Check whether this WorkOS identity already exists.
    user = (
        db.query(User)
        .filter(User.workos_user_id == workos_user_id)
        .first()
    )

    if user:
        # Never allow an email already linked to another identity.
        email_owner = (
            db.query(User)
            .filter(User.email == email, User.id != user.id)
            .first()
        )

        if email_owner:
            raise ValueError("This email is linked to another account.")

        user.email = email
        user.name = name
        db.commit()
        db.refresh(user)

        return user, False

    # Prevent silently linking an existing account by email.
    email_owner = (
        db.query(User)
        .filter(User.email == email)
        .first()
    )

    if email_owner:
        raise ValueError("This email is already linked to another account.")

    user = User(
        workos_user_id=workos_user_id,
        email=email,
        name=name,
    )

    db.add(user)

    try:
        db.commit()
        db.refresh(user)
        return user, True

    except IntegrityError:
        db.rollback()

        # Handle a concurrent request for the same WorkOS identity.
        user = (
            db.query(User)
            .filter(User.workos_user_id == workos_user_id)
            .first()
        )

        if user:
            return user, False

        raise ValueError("Could not create the finance account.")

from sqlalchemy.orm import Session
from DB.models import Expense
from datetime import date
from decimal import Decimal


def create_expense(
    db: Session,
    user_id: int,
    amount: Decimal,
    description: str,
    expense_date: date,
    category: str,
):
    new_expense = Expense(
        amount=amount,
        description=description,
        date=expense_date,
        category=category,
        user_id=user_id,
    )

    db.add(new_expense)
    db.commit()
    db.refresh(new_expense)

    return new_expense


def get_expenses(
    db: Session,
    user_id: int,
    start_date: date | None = None,
    end_date: date | None = None,
    category: str | None = None,
):
    query = db.query(Expense).filter(
        Expense.user_id == user_id
    )

    if start_date:
        query = query.filter(Expense.date >= start_date)

    if end_date:
        query = query.filter(Expense.date <= end_date)

    if category:
        query = query.filter(Expense.category == category)

    return query.order_by(Expense.date.desc()).all()


def get_expenses_paginated(
    db: Session,
    user_id: int,
    start_date: date | None = None,
    end_date: date | None = None,
    category: str | None = None,
    page: int = 1,
    limit: int = 10,
):
    query = db.query(Expense).filter(
        Expense.user_id == user_id
    )

    if start_date:
        query = query.filter(Expense.date >= start_date)

    if end_date:
        query = query.filter(Expense.date <= end_date)

    if category:
        query = query.filter(Expense.category == category)

    # Count matching records before applying pagination.
    total = query.count()

    # Fetch only the requested page.
    offset = (page - 1) * limit

    items = (
        query
        .order_by(Expense.date.desc(), Expense.id.desc())
        .offset(offset)
        .limit(limit)
        .all()
    )

    total_pages = (total + limit - 1) // limit

    return {
        "items": items,
        "total": total,
        "page": page,
        "limit": limit,
        "total_pages": total_pages,
    }


def update_expense(
    db: Session,
    user_id: int,
    expense_id: int,
    amount: Decimal | None = None,
    description: str | None = None,
    expense_date: date | None = None,
    category: str | None = None,
):
    expense = (
        db.query(Expense)
        .filter(
            Expense.id == expense_id,
            Expense.user_id == user_id,
        )
        .first()
    )

    if expense is None:
        return None

    if amount is not None:
        expense.amount = amount

    if description is not None:
        expense.description = description

    if expense_date is not None:
        expense.date = expense_date

    if category is not None:
        expense.category = category

    db.commit()
    db.refresh(expense)

    return expense



def delete_expense(
    db: Session,
    user_id: int,
    expense_id: int,
):
    expense = (
        db.query(Expense)
        .filter(
            Expense.id == expense_id,
            Expense.user_id == user_id,
        )
        .first()
    )

    if expense is None:
        return False

    db.delete(expense)
    db.commit()

    return True

def get_current_user_id(
    credentials: HTTPAuthorizationCredentials | None = Depends(security),
    db: Session = Depends(get_db_session),
) -> int:

    if credentials is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authentication required.",
        )

    if not INTERNAL_API_SECRET:
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail="API authentication is not configured.",
        )

    try:
        claims = jwt.decode(
            credentials.credentials,
            INTERNAL_API_SECRET,
            algorithms=["HS256"],
            issuer=INTERNAL_API_ISSUER,
            audience=INTERNAL_API_AUDIENCE,
            options={
                "require": ["exp", "iat", "iss", "aud", "sub"]
            },
        )

    except jwt.InvalidTokenError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired authentication token.",
        )

    workos_user_id = claims.get("sub")

    if not isinstance(workos_user_id, str) or not workos_user_id:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid user identity.",
        )

    user = (
        db.query(User)
        .filter(User.workos_user_id == workos_user_id)
        .first()
    )

    if user is None:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Finance account not found.",
        )

    return user.id

@app.post("/internal/users/provision")
def provision_user(
    payload: ProvisionUserRequest,
    x_provisioning_secret: str = Header(
        alias="X-Provisioning-Secret"
    ),
    db: Session = Depends(get_db_session),
):
    expected_secret = os.getenv("PROVISIONING_SECRET")

    if not expected_secret:
        raise HTTPException(
            status_code=503,
            detail="Provisioning is not configured.",
        )

    if not hmac.compare_digest(
        x_provisioning_secret,
        expected_secret,
    ):
        raise HTTPException(
            status_code=401,
            detail="Unauthorized.",
        )

    try:
        user, created = provision_workos_user(
            db=db,
            workos_user_id=payload.workos_user_id,
            email=payload.email,
            name=payload.name,
        )

        return {
            "message": (
                "Finance account created."
                if created
                else "Finance account already exists."
            ),
            "user_id": user.id,
            "created": created,
        }

    except ValueError as error:
        raise HTTPException(status_code=409, detail=str(error))

def get_authenticated_user_id(db) -> int:
    """
    Resolve the verified WorkOS identity to our local database user.

    Never accept a user ID supplied by an MCP client or an LLM.
    """

    access_token = get_access_token()

    if access_token is None:
        raise PermissionError("Authentication required.")

    workos_user_id = access_token.claims.get("sub")

    if not isinstance(workos_user_id, str) or not workos_user_id.strip():
        raise PermissionError("Invalid authenticated identity.")

    user = (
        db.query(User)
        .filter(User.workos_user_id == workos_user_id)
        .first()
    )

    if user is None:
        raise PermissionError(
            "Your WorkOS account is not linked to a finance account."
        )

    return user.id
